# LITTLE MARCH - a short original example of the "threetone" layout
# (Amy & Claude 2026): three audible tone voices + noise drums, no
# periodic-noise bass, so tone 3 plays a real bass line (A2 = 110 Hz is
# about the lowest tone the SN76489 can play: N = 1017).
# D major, bpm 128 with 4 steps per beat -> 7 frames per step,
# A B A B2 = 16 bars = 1792 frames = 29.9 s, loops.
title  LITTLE MARCH
prefix Lm
table  LmSoundTable
layout threetone
bpm    128 4
bar    16
floor  3
loop   on

voice Lead  tone1 atn=3 inst=p
voice Harm  tone2 atn=7 inst=pad
voice Bass  tone3 atn=4 inst=thump
voice Drums noise

inst p     pluck avals=4 astep=1 per=2 first=2 longper=3 longfrom=13
inst pad   vibrato head=14 hper=15 hfirst=15 cap=9
inst thump pluck avals=4 astep=2 per=3 first=3

drum K ctrl=2 atn=4 env=4,2,1,2 max=8     # kick: slow periodic noise, short
drum S ctrl=5 atn=4 env=5,2,2,1 max=10    # snare: white noise, medium
drum h ctrl=4 atn=7 env=3,3,1,1 max=4     # hat: white noise, fast, very short

section A bars=4
Lead:  F#5:2 A5:2 D6:4 C#6:2 B5:2 A5:4 |
       B5:3 A5:3 F#5:2 D5:8vibrato |
       G5:2 B5:2 D6:4 B5:2 G5:2 E5:4 |
       E5:3 F#5:3 G5:2 A5:8fall |
Harm:  D5:16 | D5:16 | B4:16 | C#5:16 |
Bass:  D3:2 D4 A3 D4 D3 D4 A3 D4 | B2 B3 F#3 B3 B2 B3 F#3 B3 |
       B2 G3 D3 G3 B2 G3 D3 G3 | A2 A3 E3 A3 A2 A3 E3 A3 |
Drums: K.h.S.h.K.h.S.h. | K.h.S.h.K.hKS.h. | K.h.S.h.K.h.S.h. | K.h.S.h.K.KSSSS.

section B bars=4
Lead:  B5:6slide A5:2 G5:4 D5:4 |
       C#6:6slide B5:2 A5:4 E5:4 |
       D6:3 C#6:3 B5:2 F#5:8vibrato |
       E5:2 F#5:2 G5:2 A5:2 C#6:8fall |
Harm:  D5:16 | E5:16 | D5:16 | C#5:16 |
Bass:  B2:2 G3 D3 G3 B2 G3 D3 G3 | A2 A3 E3 A3 A2 A3 E3 A3 |
       B2 B3 F#3 B3 B2 B3 F#3 B3 | A2 A3 E3 A3 A2 A3 C#3 E3 |
Drums: K.hhS.h.K.hhS.h. | K.hhS.h.K.hhS.hK | K.hhS.h.K.hhS.h. | K.hhS.hSK.SSS.S.

section B2 bars=4
Lead:  B5:6slide A5:2 G5:4 D5:4 |
       C#6:6slide B5:2 A5:4 E5:4 |
       D6:3 C#6:3 B5:2 F#5:4 A5:4 |
       G5:2 F#5:2 E5:2 C#5:2 D5:8fall |
Harm:  D5:16 | E5:16 | F#5:16 | E5:8 F#5:8 |
Bass:  B2:2 G3 D3 G3 B2 G3 D3 G3 | A2 A3 E3 A3 A2 A3 E3 A3 |
       B2 B3 F#3 B3 B2 B3 F#3 B3 | A2 A3 E3 A3 D3 D3 A2 A2 |
Drums: K.hhS.h.K.hhS.h. | K.hhS.h.K.hhS.hK | K.hhS.h.K.hhS.h. | K.h.S.h.K.SSSSSS

play A B A B2

effect Zap tone1 const=SndLmZap
  sweep 60 atn=0 vals=12 step=12 per=1 first=1 env=6,2,2,2

effect Pop noise const=SndLmPop
  noise 6 atn=0 f=12 env=6,2,2,2
